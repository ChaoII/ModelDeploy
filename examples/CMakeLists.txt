# 后端×平台 demo 矩阵：只用现有开关推导；每个 demo_<stem>_<backend>.cpp 自包含完整代码
function(md_add_demo_matrix STEM SOURCE_DIR)
  set(_base "${SOURCE_DIR}/demo_${STEM}")
  if(ENABLE_ORT)
    add_executable(demo_${STEM}_ort_cpu "${_base}_ort_cpu.cpp")
    target_link_libraries(demo_${STEM}_ort_cpu PUBLIC ${LIBRARY_NAME})
  endif()
  if(ENABLE_ORT AND WITH_GPU)
    add_executable(demo_${STEM}_ort_gpu_cuda_ep "${_base}_ort_gpu_cuda_ep.cpp")
    target_link_libraries(demo_${STEM}_ort_gpu_cuda_ep PUBLIC ${LIBRARY_NAME})
    # ORT 的 TRT EP 不需要原生 ENABLE_TRT
    add_executable(demo_${STEM}_ort_gpu_trt_ep "${_base}_ort_gpu_trt_ep.cpp")
    target_link_libraries(demo_${STEM}_ort_gpu_trt_ep PUBLIC ${LIBRARY_NAME})
  endif()
  # 原生 TRT 后端（加载预构建 .engine）
  if(ENABLE_TRT AND WITH_GPU)
    add_executable(demo_${STEM}_trt "${_base}_trt.cpp")
    target_link_libraries(demo_${STEM}_trt PUBLIC ${LIBRARY_NAME})
  endif()
  if(ENABLE_MNN)
    foreach(_s mnn_cpu mnn_opencl mnn_vulkan)
      add_executable(demo_${STEM}_${_s} "${_base}_${_s}.cpp")
      target_link_libraries(demo_${STEM}_${_s} PUBLIC ${LIBRARY_NAME})
    endforeach()
  endif()
  if(ENABLE_MNN AND WITH_GPU)
    add_executable(demo_${STEM}_mnn_cuda "${_base}_mnn_cuda.cpp")
    target_link_libraries(demo_${STEM}_mnn_cuda PUBLIC ${LIBRARY_NAME})
  endif()
  if(ENABLE_SOPHGO)
    add_executable(demo_${STEM}_sophgo_tpu_f16 "${_base}_sophgo_tpu_f16.cpp")
    target_link_libraries(demo_${STEM}_sophgo_tpu_f16 PUBLIC ${LIBRARY_NAME})
    add_executable(demo_${STEM}_sophgo_tpu_int8 "${_base}_sophgo_tpu_int8.cpp")
    target_link_libraries(demo_${STEM}_sophgo_tpu_int8 PUBLIC ${LIBRARY_NAME})
  endif()
endfunction()
