- [ ] **Step 2: 实现 draw_gpu.cu**

复用 `nv12_to_bgr.cu` 模式（thread_local workspace + caller stream）。kernel：
- 1 block per box（block 256 线程）
- 前 128 线程：填充 box 内部矩形（半透明混合：`dst = dst*(1-alpha) + color*alpha`）
- 后 128 线程：画 4 条边框 + 顶部标签条（用内嵌 8x16 ASCII 位图字体）
- 8x16 字体数据：定义 `static const uint8_t FONT8X16[128][16]`（可简化为只支持 ASCII 0x20-0x7E，用标准 8x16 字体数据）

host wrapper：
```cpp
#include "vision/common/processors/draw_gpu.cuh"
#include <cuda_runtime.h>

namespace modeldeploy::vision {
    __global__ void kernel_draw_boxes(uint8_t* bgr, int width, int height,
                                      const GpuDrawBox* boxes, int num_boxes, float alpha) {
        const int b = blockIdx.x;
        if (b >= num_boxes) return;
        const GpuDrawBox box = boxes[b];
        const int tid = threadIdx.x;
        // 0-127: fill; 128-255: border + label
        if (tid < 128) {
            // fill box interior
            const int rows = box.y2 - box.y1;
            const int cols = box.x2 - box.x1;
            const int idx = tid;
            for (int y = box.y1 + idx / 16; y < box.y2 && y < height; y += 8) {
                for (int x = box.x1 + (idx % 16); x < box.x2 && x < width; x += 16) {
                    uint8_t* p = bgr + (y * width + x) * 3;
                    p[0] = static_cast<uint8_t>(p[0] * (1.0f - alpha) + box.b * alpha);
                    p[1] = static_cast<uint8_t>(p[1] * (1.0f - alpha) + box.g * alpha);
                    p[2] = static_cast<uint8_t>(p[2] * (1.0f - alpha) + box.r * alpha);
                }
            }
        } else {
            // border: 4 lines (2px)
            const int c = tid - 128;  // 0-127 threads for border
            // ... border drawing on box perimeter ...
        }
    }

    bool draw_boxes_gpu(uint8_t* bgr, int width, int height,
                        const GpuDrawBox* d_boxes, int num_boxes,
                        float alpha, cudaStream_t stream) {
        if (!bgr || num_boxes <= 0) return true;
        bool is_internal = false;
        if (stream == nullptr) { if (cudaStreamCreate(&stream) != cudaSuccess) return false; is_internal = true; }
        dim3 grid(num_boxes);
        dim3 block(256);
        kernel_draw_boxes<<<grid, block, 0, stream>>>(bgr, width, height, d_boxes, num_boxes, alpha);
        if (is_internal) { cudaStreamSynchronize(stream); cudaStreamDestroy(stream); }
        return true;
    }
}
```
注意：完整 border/label 绘制 kernel 需在实现时给全。**字体数据来源**：使用标准 8x16 VGA 字体位图（`static const uint8_t FONT8X16[128][16]`），可从公共领域字体表获取（如 `font8x16_basic` GitHub gist），仅 ASCII 0x20-0x7E。label 渲染：每字符 8 列 × 16 行，在 box 顶部画标签背景条 + 白色/黑色文字。边框：4 条 2px 线段。实现时把 PRD §3.2 的完整方案落地（fill + border + label 三段式 block 分工）。

- [ ] **Step 2: 实现 bgr_to_nv12.cu**

BT.709 limited range 公式：
```cpp
__global__ void kernel_bgr_to_nv12_y(const uint8_t* __restrict__ bgr, uint8_t* __restrict__ y, int w, int h) {
    const int x = blockIdx.x * blockDim.x + threadIdx.x;
    const int y_ = blockIdx.y * blockDim.y + threadIdx.y;
    if (x >= w || y_ >= h) return;
    const int idx = (y_ * w + x) * 3;
    const int r = bgr[idx+2], g = bgr[idx+1], b = bgr[idx+0];
    y[y_ * w + x] = (uint8_t)(((66*r + 129*g + 25*b + 128) >> 8) + 16);
}

__global__ void kernel_bgr_to_nv12_uv(const uint8_t* __restrict__ bgr, uint8_t* __restrict__ uv, int w, int h) {
    const int x = blockIdx.x * blockDim.x + threadIdx.x;
    const int y_ = blockIdx.y * blockDim.y + threadIdx.y;
    const int uvw = w / 2, uvh = h / 2;
    if (x >= uvw || y_ >= uvh) return;
    // 2x2 平均
    int r=0,g=0,b=0;
    for (int dy=0; dy<2; ++dy) for (int dx=0; dx<2; ++dx) {
        const int idx = ((y_*2+dy)*w + (x*2+dx))*3;
        r += bgr[idx+2]; g += bgr[idx+1]; b += bgr[idx+0];
    }
    r>>=2; g>>=2; b>>=2;
    const int u = ((-38*r - 74*g + 112*b + 128) >> 8) + 128;
    const int v = ((112*r - 94*g - 18*b + 128) >> 8) + 128;
    uint8_t* p = uv + (y_*uvw + x)*2;
    p[0] = (uint8_t)u; p[1] = (uint8_t)v;
}
```
host wrapper：输入 BGR 可能是 device 或 host（`cudaPointerGetAttributes`），输出 NV12 device buffer。调用 2 个 kernel（Y 平面 + UV 平面）。
